CREATE TABLE cliente (
    cliCedula VARCHAR(15) PRIMARY KEY,
    cliNombre VARCHAR(50) NOT NULL,
    cliApellido VARCHAR(50) NOT NULL

CREATE TABLE producto (

CREATE TABLE pedido (
    pedId INT AUTO_INCREMENT PRIMARY KEY,

    FOREIGN KEY (cliCedula) REFERENCES cliente(cliCedula),

==============cruds.php==============
<?php
include ("conexion.php");
class cruds{
    public static function insertarPedido(){
        $conn = new conexion();
        $conectar = $conn->conectar();
        $cliCedula = $_POST['cliCedula'];
        $prodId = $_POST['prodId'];

        $sqlInsert = "INSERT INTO pedido (cliCedula, prodId) VALUES (:cedula, :prodId)";
        $resultado = $conectar->prepare($sqlInsert);
        $resultado->bindParam(':cedula', $cliCedula);
        $resultado->bindParam(':prodId', $prodId);
        $resultado->execute();

        $data="Pedido registrado";
        print_r(json_encode($data));
    }

    public static function listarClientes(){
        $conn = new conexion();
        $conectar = $conn->conectar();
        $sql = "SELECT cliCedula, cliNombre, cliApellido FROM cliente";
        $resultado = $conectar->prepare($sql);
        $resultado->execute();
        $data=$resultado->fetchAll(PDO::FETCH_ASSOC);
        echo json_encode($data);
    }

    public static function buscarCliente(){
        $conn = new conexion();
        $conectar = $conn->conectar();
        $cedula = $_GET['cliCedula'];
        $sql = "SELECT cliCedula, cliNombre, cliApellido FROM cliente WHERE cliCedula = :cedula";
        $resultado = $conectar->prepare($sql);
        $resultado->bindParam(':cedula', $cedula);
        $resultado->execute();
        $data=$resultado->fetchAll(PDO::FETCH_ASSOC);
        echo json_encode($data);
    }

    public static function listarProductos(){
        $conn = new conexion();
        $conectar = $conn->conectar();
        $sql = "SELECT prodId, prodNombre FROM producto";
        $resultado = $conectar->prepare($sql);
        $resultado->execute();
        $data=$resultado->fetchAll(PDO::FETCH_ASSOC);
        echo json_encode($data);
    }

    public static function pedidosCliente(){
        $conn = new conexion();
        $conectar = $conn->conectar();
        $cedula = $_GET['cliCedula'];
        $sql = "SELECT pedido.pedId, cliente.cliApellido, cliente.cliNombre, producto.prodNombre
                FROM pedido
                INNER JOIN cliente ON pedido.cliCedula = cliente.cliCedula
                INNER JOIN producto ON pedido.prodId = producto.prodId
                WHERE pedido.cliCedula = :cedula";
        $resultado = $conectar->prepare($sql);
        $resultado->bindParam(':cedula', $cedula);
        $resultado->execute();
        $data=$resultado->fetchAll(PDO::FETCH_ASSOC);
        echo json_encode($data);
    }
}
?>

        async function cargarOpciones() {
            const clientes = await fetch(ApiUrl + "?accion=listarClientes").then(r => r.json());
            const selectCli = document.getElementById("cliCedula");
            selectCli.innerHTML = "";
            clientes.forEach(c => {
                let op = document.createElement("option");
                op.value = c.cliCedula;
                op.text = c.cliApellido + " " + c.cliNombre;
                selectCli.appendChild(op);
            });

            const productos = await fetch(ApiUrl + "?accion=listarProductos").then(r => r.json());
            const selectProd = document.getElementById("prodId");
            selectProd.innerHTML = "";
            productos.forEach(p => {

                op.text = p.prodNombre;
                selectProd.appendChild(op);
            });
        }

        document.getElementById("btnRegistrar").addEventListener("click", async () => {
            const datos = new FormData();
